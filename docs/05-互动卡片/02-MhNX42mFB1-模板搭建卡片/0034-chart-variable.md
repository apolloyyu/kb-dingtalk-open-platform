# 图表变量

本文介绍了图表变量的数据协议，同时也为图表组件支持的折线图、柱状图和饼状图提供了示例。

## **概述**

图表变量是「图表」组件消费的变量类型，结合图表变量，「图表」组件可以展示折线图、柱状图和饼状图三种图表类型。

## **变量数据协议**

```
enum ChartType {
  /** 折线图 */
  Line = "lineChart",
  /** 柱状图 */
  Histogram = "histogram",
  /** 饼状图 */
  Pie = "pieChart",
}

/**
 * 度量 scale
 * 完整配置请参考：https://g2-v4.antv.vision/zh/docs/api/general/scale
 */
interface AxisConfig {
  /** 坐标轴上刻度点的个数，不同的度量类型对应不同的默认值 */
  tickCount?: number;
  /** 指定不同的度量类型，支持的 type 为 identity、linear、cat、timeCat */
  type?: string;
  /** 输出数据的范围，数值类型的默认值为 [0, 1]，格式为 [min, max]，min 和 max 均为 0 至 1 范围的数据 */
  range?: [number, number];
  /** 该数据字段的显示别名，一般用于将字段的英文名称转换成中文名 */
  alias?: string;
  /** 用于指定坐标轴上刻度点的文本信息，当用户设置了 ticks 就会按照 ticks 的个数和文本来显示 */
  ticks: number[];
}

interface LineChartData {
  /** 横坐标 */
  x: string;
  /** 纵坐标，必须是数字 */
  y: number;
  /** 分组字段 */
  type?: string;
}

interface BaseConfig {
  /** 线条颜色 */
  color?: string | string[];
  /** X 轴的度量设置，透传给 G2 */
  xAxisConfig?: AxisConfig;
  /** Y 轴的度量设置，透传给 G2 */
  yAxisConfig?: AxisConfig;
  /** 
   * X 轴的线条配置，透传给 G2
   * AxisOption 参考 https://g2-v4.antv.vision/zh/docs/api/general/axis
   */
  xAxisOptions?: AxisOption;
  /** Y 轴的线条配置，透传给 G2 */
  yAxisOptions?: AxisOption;
  /** 图表的边距 */
  padding?: [number, number, number, number];
  /** 
   * 图例样式，透传给 G2
   * LegendOption 参考 https://g2-v4.antv.vision/zh/docs/api/general/legend
   */
  legend?: LegendOption;
}

/** 折线图配置 */
interface LineChartConfig extends BaseConfig {
  /** 图表是否要旋转 */
  rotate?: boolean;
  /** 线条的形状，直接设置给 G2 */
  lineShape?: 'line' | 'dot' | 'smooth';
}

interface HistogramChartData {
  x: string;
  y: number | [number, number];
  /** 分组字段 */
  type?: string;
}

interface HistogramChartConfig extends LineChartConfig {
  /** 是否层叠柱状图，传 false 或不传默认分组显示 */
  stack?: boolean;
}

interface PieChartData {
  x: string;
  y: number;
}

type PieChartConfig = BaseConfig & {
  pieChartStyle?: {
  legend: 'default' | 'percentage' | 'num';
};
};

/** 折线图协议 */
interface LineChartProtocol {
  type: ChartType.Line;
  /** 图表数据 */
  data: LineChartData[];
  /** 图表配置 */
  config?: LineChartConfig;
}

/** 柱状图协议 */
interface HistogramChartProtocol {
  type: ChartType.Histogram;
  /** 图表数据 */
  data: HistogramChartData[];
  /** 图表配置 */
  config?: HistogramChartConfig;
}

/** 饼状图 */
interface PieChartProtocol {
  type: ChartType.Pie;
  /** 图表数据 */
  data: PieChartData[];
  /** 图表配置 */
  config?: PieChartConfig;
}

/** 图表变量协议 */
type ChartDataProtocol =
  | LineChartProtocol
  | HistogramChartProtocol
  | PieChartProtocol;
```

## **折线图**

### **协议定义**

```
/** 折线图协议 */
interface LineChartProtocol {
  type: ChartType.Line;
  /** 图表数据 */
  data: LineChartData[];
  /** 图表配置 */
  config?: LineChartConfig;
}
```
